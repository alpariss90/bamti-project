Voici le récapitulatif exact du comportement actuellement implémenté :

Table service_person reçoit une nouvelle ligne dans exactement 4 cas :

1. Création d'un personnel (formulaire "Ajouter")
Si le champ Date prise de service est rempli → insertion d'une ligne :
type = "prise"
date_service = la date saisie
motif = "" (vide)
Si le champ est laissé vide → aucune ligne n'est créée.
2. Modification d'un personnel (formulaire "Modifier")
Le champ Date prise de service n'est visible en modification que si il était vide avant.
S'il était vide et qu'on le renseigne maintenant → insertion d'une ligne type = "prise", date_service = date saisie, motif = "".
S'il était déjà renseigné avant → le champ est caché, donc rien n'est modifié et aucune ligne n'est créée.
3. Bouton "Fin service" (uniquement visible si le personnel est actif)
Ouvre le modal → l'utilisateur saisit une date et un motif obligatoires.
À l'enregistrement :
insertion d'une ligne type = "arret", date_service = date saisie, motif = motif saisi
etat du personnel passe à false (inactif)
4. Bouton "Reprise service" (uniquement visible si le personnel est inactif)
Ouvre le même modal (libellés adaptés) → date et motif obligatoires.
À l'enregistrement :
insertion d'une ligne type = "prise", date_service = date saisie, motif = motif saisi
etat du personnel passe à true (actif)
Résumé type :

"prise" → soit automatique (création/1ère modif avec la date de prise de service, motif vide), soit manuelle via "Reprise service" (avec motif).
"arret" → uniquement via "Fin service" (avec motif obligatoire).

-- ===== Initialisation du stock sachets à partir des ventes déjà existantes =====
-- À exécuter UNE SEULE FOIS, juste après la création des 2 tables ci-dessus et
-- avant de commencer à utiliser le module (sinon les quantités seront comptées
-- deux fois). Système en place depuis plus d'un an : on reconstitue l'historique
-- mois par mois plutôt qu'en un seul bloc — une ligne par mois calendaire, datée
-- au 1er du mois concerné.
-- Sortie = somme des ventes du mois (table `ventes`).
-- Entrée = somme des ventes annulées du mois (table `ventes_annuler`), en restitution.

INSERT INTO `stock_sachet_sorties` (`quantite`, `date_sortie`, `motif`, `id_vente`, `createdBy`, `createdAt`, `updatedAt`)
SELECT
  SUM(`quantite`),
  STR_TO_DATE(CONCAT(YEAR(`date_vente`), '-', MONTH(`date_vente`), '-01'), '%Y-%m-%d'),
  CONCAT('Initialisation - ventes de ', DATE_FORMAT(`date_vente`, '%m/%Y')),
  NULL,
  'system',
  NOW(),
  NOW()
FROM `ventes`
GROUP BY YEAR(`date_vente`), MONTH(`date_vente`);

INSERT INTO `stock_sachet_entrees` (`quantite`, `date_production`, `observation`, `createdBy`, `createdAt`, `updatedAt`)
SELECT
  SUM(`quantite`),
  STR_TO_DATE(CONCAT(YEAR(`date_vente`), '-', MONTH(`date_vente`), '-01'), '%Y-%m-%d'),
  CONCAT('Initialisation - restitution ventes annulées de ', DATE_FORMAT(`date_vente`, '%m/%Y')),
  'system',
  NOW(),
  NOW()
FROM `ventes_annuler`
GROUP BY YEAR(`date_vente`), MONTH(`date_vente`);


-- ===== Régularisation : ramener le stock exactement à 0 =====
-- Les scripts ci-dessus enregistrent des SORTIES historiques (les ventes) et déjà
-- un peu d'ENTRÉE (restitution des ventes annulées) -> stock encore négatif.
-- Pour tomber pile à 0, chaque entrée de régularisation ne compense QUE la
-- différence (sortie du mois − restitution déjà enregistrée ce même mois), pas
-- toute la sortie : sortie_mois = entrée_restitution_mois + entrée_régularisation_mois.
-- Le WHERE/JOIN ne touche que les lignes créées par les scripts d'initialisation
-- ci-dessus (jamais les vraies sorties/entrées enregistrées depuis la mise en service).

INSERT INTO `stock_sachet_entrees` (`quantite`, `date_production`, `observation`, `createdBy`, `createdAt`, `updatedAt`)
SELECT
  s.`quantite` - COALESCE(r.`quantite`, 0),
  DATE_ADD(s.`date_sortie`, INTERVAL 1 DAY),
  CONCAT('Régularisation stock - production non enregistrée de ', DATE_FORMAT(s.`date_sortie`, '%m/%Y')),
  'system',
  NOW(),
  NOW()
FROM `stock_sachet_sorties` s
LEFT JOIN `stock_sachet_entrees` r
  ON r.`date_production` = s.`date_sortie`
  AND r.`observation` LIKE 'Initialisation - restitution%'
WHERE s.`motif` LIKE 'Initialisation - ventes de%'
  AND (s.`quantite` - COALESCE(r.`quantite`, 0)) > 0;

CREATE OR REPLACE VIEW `vsalaires` AS
SELECT
  `s`.`id` AS `id`,
  `s`.`id_personnel` AS `id_personnel`,
  `p`.`nom` AS `nom`,
  `p`.`prenom` AS `prenom`,
  `p`.`telephone` AS `telephone`,
  `s`.`mois` AS `mois`,
  `s`.`annee` AS `annee`,
  `s`.`salaire_base` AS `salaire_base`,
  `s`.`montant_retenue` AS `montant_retenue`,
  `s`.`montant_jours_non_payes` AS `montant_jours_non_payes`,
  `s`.`montant_salaire` AS `montant_salaire`,
  `s`.`montant_credit` AS `montant_credit`,
  `s`.`montant_gratification` AS `montant_gratification`,
  (`s`.`montant_salaire` + `s`.`montant_credit` + `s`.`montant_gratification`) AS `montant_reçu`,
  `ss`.`ms` AS `ms`,
  `ss`.`mc` AS `mc`,
  `ss`.`mg` AS `mg`,
  `ss`.`mr` AS `mr`
FROM `salaires` `s`
JOIN (
  SELECT `mois`, `annee`,
         SUM(`montant_salaire`) AS `ms`,
         SUM(`montant_credit`) AS `mc`,
         SUM(`montant_gratification`) AS `mg`,
         SUM(`montant_salaire` + `montant_credit` + `montant_gratification`) AS `mr`
  FROM `salaires`
  GROUP BY `mois`, `annee`
) `ss` ON `s`.`mois` = `ss`.`mois` AND `s`.`annee` = `ss`.`annee`
JOIN `personnels` `p` ON `p`.`id` = `s`.`id_personnel`;

-- ÉTAPE 1 : nouvelle colonne (vide = encaisseur inconnu, pour les anciens versements)
ALTER TABLE `paiements`
  ADD COLUMN `id_user` int(11) DEFAULT NULL AFTER `id_vente`,
  ADD INDEX `idx_paiements_user_date` (`id_user`, `date`);

-- ÉTAPE 2 : historique, uniquement les cas sûrs

-- 2a. Paiements créés en même temps que la vente (moins de 5 s d'écart) : encaissés par le vendeur
-- Contrôle (nombre de lignes qui seront remplies) :
SELECT COUNT(*) AS paiements_avec_la_vente
FROM `paiements` p
JOIN `ventes` v ON v.`id` = p.`id_vente`
WHERE p.`id_user` IS NULL
  AND ABS(TIMESTAMPDIFF(SECOND, v.`createdAt`, p.`createdAt`)) <= 5;

UPDATE `paiements` p
JOIN `ventes` v ON v.`id` = p.`id_vente`
SET p.`id_user` = v.`user`
WHERE p.`id_user` IS NULL
  AND ABS(TIMESTAMPDIFF(SECOND, v.`createdAt`, p.`createdAt`)) <= 5;

-- 2b. Ventes des revendeurs : le revendeur encaisse toujours lui-même ses ventes
-- Contrôle :
SELECT COUNT(*) AS paiements_revendeurs
FROM `paiements` p
JOIN `ventes` v ON v.`id` = p.`id_vente`
JOIN `users` u ON u.`id` = v.`user`
WHERE p.`id_user` IS NULL
  AND u.`profil` = 'revendeur';

UPDATE `paiements` p
JOIN `ventes` v ON v.`id` = p.`id_vente`
JOIN `users` u ON u.`id` = v.`user`
SET p.`id_user` = v.`user`
WHERE p.`id_user` IS NULL
  AND u.`profil` = 'revendeur';

-- Bilan : paiements dont l'encaisseur reste inconnu (anciens versements ultérieurs)
SELECT
  SUM(`id_user` IS NOT NULL) AS encaisseur_connu,
  SUM(`id_user` IS NULL)     AS encaisseur_inconnu
FROM `paiements`;

-- Une ligne par personne (caissier / gestionnaire) et par date clôturée
CREATE TABLE IF NOT EXISTS `caisse_journees` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `id_user` int(11) NOT NULL,                 -- qui a encaissé
  `date` date NOT NULL,                       -- jour clôturé
  `montant_encaisse` double NOT NULL,         -- total encaissé ce jour, figé à la clôture
  `id_detenteur` int(11) NOT NULL,            -- chez qui est l'argent actuellement
  `montant_detenu` double NOT NULL,           -- combien il en détient
  `createdAt` datetime NOT NULL,
  `updatedAt` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_caisse_journee_user_date` (`id_user`, `date`),
  KEY `idx_caisse_journee_detenteur` (`id_detenteur`),
  KEY `idx_caisse_journee_date` (`date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Un envoi d'argent d'une personne à une autre (peut regrouper plusieurs journées)
CREATE TABLE IF NOT EXISTS `caisse_remises` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `id_expediteur` int(11) NOT NULL,
  `id_destinataire` int(11) NOT NULL,
  `statut` enum('en_attente','recue','refusee','annulee') NOT NULL DEFAULT 'en_attente',
  `commentaire_refus` text DEFAULT NULL,
  `date_reponse` datetime DEFAULT NULL,       -- reçue, refusée ou annulée le
  `createdAt` datetime NOT NULL,
  `updatedAt` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_caisse_remise_destinataire` (`id_destinataire`, `statut`),
  KEY `idx_caisse_remise_expediteur` (`id_expediteur`, `statut`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Une ligne par journée dans une remise
CREATE TABLE IF NOT EXISTS `caisse_remise_lignes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `id_remise` int(11) NOT NULL,
  `id_journee` int(11) NOT NULL,
  `montant_attendu` double NOT NULL,          -- ce que l'expéditeur détenait pour cette journée
  `montant_remis` double NOT NULL,            -- ce qu'il remet (au plus montant_attendu)
  `commentaire_ecart` text DEFAULT NULL,      -- obligatoire si montant_remis < montant_attendu
  `createdAt` datetime NOT NULL,
  `updatedAt` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_caisse_ligne_remise` (`id_remise`),
  KEY `idx_caisse_ligne_journee` (`id_journee`),
  CONSTRAINT `fk_caisse_ligne_remise` FOREIGN KEY (`id_remise`) REFERENCES `caisse_remises` (`id`),
  CONSTRAINT `fk_caisse_ligne_journee` FOREIGN KEY (`id_journee`) REFERENCES `caisse_journees` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- BUT : savoir à tout moment chez qui se trouve l'argent cash encaissé par un caissier un jour donné,
--       de main en main jusqu'à l'admin, avec les écarts (manquants) et leurs explications.

-- PRÉREQUIS : la colonne paiements.id_user (section « Paiements : qui a encaissé » plus haut).
--   Chaque paiement enregistre qui l'a encaissé ; le montant d'une journée = somme des paiements

--
-- DÉPLOIEMENT (dans cet ordre) :
--   1. Exécuter la section « Paiements : qui a encaissé » (étape 1, puis étape 2 après les SELECT de contrôle)
--   2. Exécuter la section « Clôture de caisse » (3 tables caisse_*)
--   3. Vérifier DATE_DEBUT (voir RÉGLAGES), puis déployer le code et redémarrer

--   6. Suivi — menu Caisse > Suivi : par période et par caissier, chez qui est l'argent, combien,

--      Caissier : ses journées ; gestionnaire : les siennes + celles qu'il détient ou qui sont passées par lui ; admin : tout.
--
-- BLOCAGE : une fois une date clôturée, son encaisseur (caissier/gestionnaire) ne peut plus enregistrer
