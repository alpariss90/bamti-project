create view vpersonnel_avance as SELECT id_personnel, mois, annee, type, sum(montant) as montant FROM `personnel_avances` group by id_personnel, mois, annee, type


update salaires s join vpersonnel_avance vp on vp.id_personnel=s.id_personnel set s.montant_credit=vp.montant where vp.type='credit'
update salaires s join vpersonnel_avance vp on vp.id_personnel=s.id_personnel set s.montant_gratification=vp.montant where vp.type='gratification'


create view vsalaires as SELECT
s.id, s.id_personnel, p.nom, p.prenom, p.telephone, s.mois, s.annee, s.montant_salaire, s.montant_credit, s.montant_gratification, (s.montant_salaire + s.montant_credit + s.montant_gratification) as montant_reçu, ss.ms, ss.mc, ss.mg, ss.mr
FROM
salaires s
JOIN
(select ss.mois, ss.annee, sum(ss.montant_salaire) as ms, sum(ss.montant_credit) as mc, sum(ss.montant_gratification) as mg, sum(ss.montant_salaire + ss.montant_credit + ss.montant_gratification) as mr from salaires ss group by ss.mois, ss.annee) ss
on s.mois=ss.mois and s.annee=ss.annee
join (select * from personnels) p on p.id=s.id_personnel
